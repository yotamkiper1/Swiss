/// Handwriting-style fonts for the notebook (chosen in its Aa menu). Chalkboard is the default.
enum NotebookFont: String, CaseIterable, Identifiable {
    case chalkboard, bradley, noteworthy, markerFelt, rounded, plain
    static let key = "notebook.font"
    var id: String { rawValue }
    var title: String {
        switch self {
        case .chalkboard: "Chalkboard"
        case .bradley: "Bradley Hand"
        case .noteworthy: "Noteworthy"
        case .markerFelt: "Marker Felt"
        case .rounded: "Rounded"
        case .plain: "Plain"
        }
    }
    var nsFont: NSFont {
        let size: CGFloat = 17
        switch self {
        case .chalkboard: return NSFont(name: "ChalkboardSE-Light", size: size) ?? .systemFont(ofSize: size)
        case .bradley: return NSFont(name: "BradleyHandITCTT-Bold", size: size + 2) ?? .systemFont(ofSize: size)
        case .noteworthy: return NSFont(name: "Noteworthy-Light", size: size + 1) ?? .systemFont(ofSize: size)
        case .markerFelt: return NSFont(name: "MarkerFelt-Thin", size: size + 1) ?? .systemFont(ofSize: size)
        case .rounded:
            let base = NSFont.systemFont(ofSize: size)
            return base.fontDescriptor.withDesign(.rounded).flatMap { NSFont(descriptor: $0, size: size) } ?? base
        case .plain: return .systemFont(ofSize: size)
        }
    }
    static var current: NotebookFont { NotebookFont(rawValue: UserDefaults.standard.string(forKey: key) ?? "") ?? .chalkboard }
}

/// Hebrew writing gets a Hebrew font of a similar feel (Corsiva Hebrew is handwritten-looking).
enum HebrewFont: String, CaseIterable, Identifiable {
    case corsiva, raanana, arial
    static let key = "notebook.hebrewFont"
    var id: String { rawValue }
    var title: String {
        switch self {
        case .corsiva: "Corsiva Hebrew"
        case .raanana: "Raanana"
        case .arial: "Arial Hebrew"
        }
    }
    var nsFont: NSFont {
        switch self {
        case .corsiva: NSFont(name: "CorsivaHebrew", size: 22) ?? .systemFont(ofSize: 17)
        case .raanana: NSFont(name: "Raanana", size: 19) ?? .systemFont(ofSize: 17)
        case .arial: NSFont(name: "ArialHebrew-Light", size: 18) ?? .systemFont(ofSize: 17)
        }
    }
    static var current: HebrewFont { HebrewFont(rawValue: UserDefaults.standard.string(forKey: key) ?? "") ?? .corsiva }
}

enum NotebookStyle {
    /// Ruled lines are this far apart.
    static let lineHeight: CGFloat = 30
    /// Where the first ruled line is (the title sits above it).
    static let topMargin: CGFloat = 114
    /// A note's stored `y` is this far above the line its first row rests on.
    static let baselineOffset: CGFloat = 26
    static let paper = Color(white: 0.99)
    static let rule = Color(white: 0.0).opacity(0.055)
    static let ink = Color(white: 0.16)
    static let quoteInk = Color(white: 0.42)

    static func marginX(compact: Bool) -> CGFloat { compact ? 18 : 28 }

    /// The top of a note whose first line rests on the ruled line nearest `y`.
    static func snappedTop(_ y: CGFloat) -> CGFloat {
        let line = max(0, ((y + baselineOffset - topMargin) / lineHeight).rounded())
        return topMargin + line * lineHeight - baselineOffset
    }

    static func isHebrew(_ text: String) -> Bool {
        text.unicodeScalars.contains { (0x0590...0x05FF).contains($0.value) }
    }

    /// The font a note is written in: Hebrew text gets the Hebrew font.
    static func font(for text: String) -> NSFont {
        isHebrew(text) ? HebrewFont.current.nsFont : NotebookFont.current.nsFont
    }

    /// How far down from its stored `y` the text box goes, so the writing sits just on the line.
    /// (Each row is one line tall and the letters sit at the bottom of it.)
    static func textOffset(_ font: NSFont) -> CGFloat {
        baselineOffset - 3 - lineHeight + abs(font.descender)
    }

    /// Extra space between rows so every row lands on the next ruled line.
    static func rowSpacing(_ font: NSFont) -> CGFloat {
        max(0, lineHeight - NSLayoutManager().defaultLineHeight(for: font))
    }

    /// How many ruled lines a note takes up.
    static func rows(_ text: String, width: CGFloat) -> Int {
        let font = font(for: text)
        let height = ((text.isEmpty ? " " : text) as NSString).boundingRect(
            with: CGSize(width: width, height: .greatestFiniteMagnitude),
            options: [.usesLineFragmentOrigin], attributes: [.font: font]).height
        return max(1, Int((height / NSLayoutManager().defaultLineHeight(for: font)).rounded(.up)))
    }
}

// MARK: - Store

extension Store {
    /// This project's pages, oldest first.
    var notebookPages: [NotebookPage] {
        pages.filter { $0.projectID == currentPlace.project }.sorted { $0.createdAt < $1.createdAt }
    }

    var currentPage: NotebookPage? {
        let list = notebookPages
        return list.first { $0.id == notebookPageID } ?? list.first
    }

    private func pageIndex(_ id: UUID) -> Int? { pages.firstIndex { $0.id == id } }

    private func noteLocation(_ noteID: UUID) -> (page: Int, note: Int)? {
        for (p, page) in pages.enumerated() {
            if let n = page.notes.firstIndex(where: { $0.id == noteID }) { return (p, n) }
        }
        return nil
    }

    func note(_ id: UUID) -> PageNote? { noteLocation(id).map { pages[$0.page].notes[$0.note] } }

    /// The current page, made if this project doesn't have one yet (the default page you start with).
    @discardableResult
    func ensurePage() -> UUID {
        if let page = currentPage { return page.id }
        let page = NotebookPage(projectID: currentPlace.project, title: "Notes")
        pages.append(page)
        notebookPageID = page.id
        save()
        return page.id
    }

    /// Click on the page: start a note there, resting on the nearest line.
    func addNote(at point: CGPoint, pageWidth: CGFloat, text: String = "") -> UUID {
        endEditing()
        checkpoint("Write")
        let pageID = ensurePage()
        let x = min(max(point.x - 4, 8), pageWidth - 90)
        let note = PageNote(x: x, y: NotebookStyle.snappedTop(point.y), width: min(360, pageWidth - x - 14), text: text)
        if let index = pageIndex(pageID) { pages[index].notes.append(note) }
        editingNote = (note.id, "")
        editingCheckpoint = undoStack.count
        save()
        return note.id
    }

    /// Focus moved into a note: remember it so the whole edit is one undo step.
    func beginEditing(_ id: UUID) {
        guard editingNote?.id != id else { return }
        endEditing()
        checkpoint("Write")
        editingCheckpoint = undoStack.count
        editingNote = (id, note(id)?.text ?? "")
    }

    /// Focus left a note: empty notes disappear, and an edit that changed nothing leaves no undo step.
    func endEditing() {
        guard let editing = editingNote else { return }
        editingNote = nil
        let current = note(editing.id)
        let text = current?.text.trimmingCharacters(in: .whitespacesAndNewlines) ?? ""
        if text.isEmpty, let loc = noteLocation(editing.id) {
            pages[loc.page].notes.remove(at: loc.note)
        }
        let unchanged = (current?.text ?? "") == editing.original || (text.isEmpty && editing.original.isEmpty)
        if unchanged, undoStack.count == editingCheckpoint { undoStack.removeLast() }
        save()
    }

    func setNoteText(_ id: UUID, _ text: String) {
        guard let loc = noteLocation(id) else { return }
        var note = pages[loc.page].notes[loc.note]
        // Typing "> " at the start turns the note into a quote.
        if text.hasPrefix("> ") {
            note.isQuote = true
            note.text = String(text.dropFirst(2))
        } else {
            note.text = text
        }
        pages[loc.page].notes[loc.note] = note
        save()
    }

    func setPageTitle(_ id: UUID, _ title: String) {
        guard let index = pageIndex(id) else { return }
        pages[index].title = title
        save()
    }

        func moveNote(_ id: UUID, to point: CGPoint, pageWidth: CGFloat) {
        guard let loc = noteLocation(id) else { return }
        checkpoint("Move Note")
        var note = pages[loc.page].notes[loc.note]
        note.x = min(max(point.x, 8), pageWidth - 90)
        note.y = NotebookStyle.snappedTop(point.y)
        note.width = min(note.width, pageWidth - note.x - 14)
        pages[loc.page].notes[loc.note] = note
        save()
    }

    func toggleQuote(_ id: UUID) {
        guard let loc = noteLocation(id) else { return }
        checkpoint("Quote")
        pages[loc.page].notes[loc.note].isQuote.toggle()
        save()
    }

    func deleteNote(_ id: UUID) {
        guard let loc = noteLocation(id) else { return }
        if editingNote?.id == id { editingNote = nil }
        checkpoint("Delete Note")
        _ = withAnimation(.easeOut(duration: 0.15)) { pages[loc.page].notes.remove(at: loc.note) }
        save()
    }

    func newPage() {
        endEditing()
        checkpoint("New Page")
        let page = NotebookPage(projectID: currentPlace.project, title: "Untitled")
        pages.append(page)
        notebookPageID = page.id
        save()
    }

    func deleteCurrentPage() {
        guard let page = currentPage else { return }
        endEditing()
        checkpoint("Delete Page")
        let list = notebookPages
        let index = list.firstIndex { $0.id == page.id } ?? 0
        pages.removeAll { $0.id == page.id }
        let remaining = notebookPages
        notebookPageID = remaining.isEmpty ? nil : remaining[min(index, remaining.count - 1)].id
        save()
        flash("Deleted page")
    }

    func stepPage(_ offset: Int) {
        let list = notebookPages
        guard !list.isEmpty else { return }
        endEditing()
        let index = list.firstIndex { $0.id == currentPage?.id } ?? 0
        notebookPageID = list[min(max(index + offset, 0), list.count - 1)].id
    }

    /// ⌘V on the notebook (not while typing): the copied text becomes a note under the others.
    func pasteIntoNotebook(pageWidth: CGFloat, compact: Bool) {
        guard let text = NSPasteboard.general.string(forType: .string)?.trimmingCharacters(in: .whitespacesAndNewlines),
              !text.isEmpty else {
            NSSound.beep()
            return
        }
        let bottom = currentPage?.notes.map { $0.y + CGFloat(NotebookStyle.rows($0.text, width: $0.width)) * NotebookStyle.lineHeight }.max()
            ?? NotebookStyle.topMargin
        _ = addNote(at: CGPoint(x: NotebookStyle.marginX(compact: compact) + 10, y: bottom), pageWidth: pageWidth, text: text)
        endEditing()
        clipboardSeenCount = NSPasteboard.general.changeCount
    }
}

/// 📓 in the header: switches between the board and the notebook.
struct NotebookButton: View {
    @EnvironmentObject var store: Store

    var body: some View {
        Button {
            store.endEditing()
            withAnimation(.easeOut(duration: 0.2)) { store.showingNotebook.toggle() }
        } label: {
            Image(systemName: store.showingNotebook ? "book.fill" : "book")
                .font(.system(size: 12))
                .frame(width: 24, height: 24)
                .background(store.showingNotebook ? Color.primary.opacity(0.09) : .clear, in: RoundedRectangle(cornerRadius: 6))
                .contentShape(Rectangle())
        }
        .buttonStyle(QuietButtonStyle())
        .foregroundStyle(store.showingNotebook ? .primary : .secondary)
        .help(store.showingNotebook ? "Back to the board" : "Notebook: lined pages to write on")
    }
}

/// Lined pages. Click anywhere to write there. Return goes down a line in the same place,
/// Tab starts the next line at the left edge of the page.
struct NotebookView: View {
    @EnvironmentObject var store: Store
    let compact: Bool
    @FocusState private var focused: UUID?
    @AppStorage(NotebookFont.key) private var fontChoice = NotebookFont.chalkboard.rawValue
    @AppStorage(HebrewFont.key) private var hebrewChoice = HebrewFont.corsiva.rawValue

    var body: some View {
        let page = store.currentPage
        let pages = store.notebookPages
        VStack(spacing: 0) {
            header(page: page, pages: pages)
            GeometryReader { geo in
                let width = geo.size.width - 24
                ScrollView {
                    paper(page: page, width: width, minHeight: geo.size.height - 16)
                        .padding(.horizontal, 12)
                        .padding(.bottom, 12)
                }
                .background(
                    // Remembers the page width for ⌘V pastes.
                    Color.clear.onAppear { store.notebookWidth[compact] = width }
                        .onChange(of: width) { store.notebookWidth[compact] = width }
                )
            }
        }
        .onAppear { store.ensurePage() }
        .onChange(of: focused) {
            if focused != nil { store.endEditing() } // into the title
        }
        .onDisappear { store.endEditing() }
    }

    private func header(page: NotebookPage?, pages: [NotebookPage]) -> some View {
        let index = page.flatMap { p in pages.firstIndex { $0.id == p.id } }
        return HStack(spacing: 4) {
            if pages.count > 1 {
                Button { store.stepPage(-1) } label: { Image(systemName: "chevron.left").frame(width: 20, height: 20) }
                    .buttonStyle(QuietButtonStyle())
                    .disabled(index == 0)
                    .keyboardShortcut("[", modifiers: [.command, .shift])
                Text("\((index ?? 0) + 1) / \(pages.count)")
                    .font(.system(size: 11))
                    .monospacedDigit()
                Button { store.stepPage(1) } label: { Image(systemName: "chevron.right").frame(width: 20, height: 20) }
                    .buttonStyle(QuietButtonStyle())
                    .disabled(index == pages.count - 1)
                    .keyboardShortcut("]", modifiers: [.command, .shift])
            }
            Spacer()
            Menu {
                Section("Writing") {
                    ForEach(NotebookFont.allCases) { font in
                        Button { fontChoice = font.rawValue } label: {
                            if font.rawValue == fontChoice { Label(font.title, systemImage: "checkmark") } else { Text(font.title) }
                        }
                    }
                }
                Section("Hebrew") {
                    ForEach(HebrewFont.allCases) { font in
                        Button { hebrewChoice = font.rawValue } label: {
                            if font.rawValue == hebrewChoice { Label(font.title, systemImage: "checkmark") } else { Text(font.title) }
                        }
                    }
                }
            } label: {
                Text("Aa").font(.system(size: 12, weight: .medium))
            }
            .menuStyle(.borderlessButton)
            .fixedSize()
            .help("Notebook font")
            Menu {
                Button("New Page") { store.newPage() }
                Button("Delete This Page", role: .destructive) { store.deleteCurrentPage() }
                    .disabled(page == nil)
            } label: {
                Image(systemName: "plus").font(.system(size: 11))
            }
            .menuStyle(.borderlessButton)
            .fixedSize()
            .help("Pages")
        }
        .foregroundStyle(.secondary)
        .padding(.horizontal, 14)
        .padding(.vertical, 5)
    }

    private func paper(page: NotebookPage?, width: CGFloat, minHeight: CGFloat) -> some View {
        let notes = page?.notes ?? []
        let lowest = notes.map { $0.y + CGFloat(NotebookStyle.rows($0.text, width: $0.width)) * NotebookStyle.lineHeight }.max() ?? 0
        let height = max(minHeight, lowest + NotebookStyle.lineHeight * 6)
        let marginX = NotebookStyle.marginX(compact: compact)
        _ = fontChoice + hebrewChoice // redraw when the font changes
        return ZStack(alignment: .topLeading) {
            Canvas { context, size in
                var y = NotebookStyle.topMargin
                while y < size.height {
                    context.fill(Path(CGRect(x: marginX, y: y, width: size.width - marginX * 2, height: 1)), with: .color(NotebookStyle.rule))
                    y += NotebookStyle.lineHeight
                }
            }
            .background(NotebookStyle.paper)
            .contentShape(Rectangle())
            .onTapGesture(coordinateSpace: .local) { location in
                guard location.y > NotebookStyle.topMargin - NotebookStyle.lineHeight else { return } // the title area
                focused = nil
                store.focusNoteID = store.addNote(at: location, pageWidth: width)
            }

            if let page { title(page, marginX: marginX, width: width) }

            if notes.isEmpty {
                Text("Click anywhere to write · Return goes down a line · Tab starts at the left")
                    .font(.system(size: 12))
                    .foregroundStyle(NotebookStyle.ink.opacity(0.28))
                    .offset(x: marginX, y: NotebookStyle.topMargin + 8)
                    .allowsHitTesting(false)
            }

            ForEach(notes) { note in
                NoteOnPage(note: note, pageWidth: width, marginX: marginX)
                    .id(note.id.uuidString + fontChoice + hebrewChoice)
            }
        }
        .frame(width: width, height: height)
        .clipShape(RoundedRectangle(cornerRadius: 6, style: .continuous))
        .coordinateSpace(name: "page")
        .environment(\.colorScheme, .light) // paper stays paper in dark mode
    }

    /// The page's title (in the notebook's handwriting, bigger) and the date it was started.
    private func title(_ page: NotebookPage, marginX: CGFloat, width: CGFloat) -> some View {
        let text = page.title ?? ""
        let base = NotebookStyle.font(for: text)
        let font = NSFontManager.shared.convert(base, toSize: base.pointSize * 1.55)
        return VStack(alignment: .leading, spacing: 2) {
            TextField("Title", text: Binding(get: { page.title ?? "" }, set: { store.setPageTitle(page.id, $0) }))
                .textFieldStyle(.plain)
                .font(Font(font))
                .foregroundStyle(NotebookStyle.ink)
                .focused($focused, equals: page.id)
                .onSubmit { focused = nil }
            Text(page.createdAt, format: .dateTime.day().month(.wide).year())
                .font(.system(size: 11))
                .foregroundStyle(NotebookStyle.ink.opacity(0.4))
        }
        .frame(width: width - marginX * 2, alignment: .leading)
        .offset(x: marginX, y: 22)
    }
}

/// One piece of writing on the page. Drag the ≡ handle to move it, right-click for quote or delete.
private struct NoteOnPage: View {
    @EnvironmentObject var store: Store
    let note: PageNote
    let pageWidth: CGFloat
    let marginX: CGFloat
    @State private var drag: CGSize = .zero
    @State private var hovering = false

    var body: some View {
        let isFocused = store.editingNote?.id == note.id
        let font = NotebookStyle.font(for: note.text)
        HStack(alignment: .top, spacing: 6) {
            if note.isQuote {
                Capsule()
                    .fill(NotebookStyle.quoteInk.opacity(0.35))
                    .frame(width: 2)
                    .padding(.vertical, 6)
            }
            NotebookTextView(
                text: note.text,
                font: font,
                color: note.isQuote ? NSColor(white: 0.42, alpha: 1) : NSColor(white: 0.16, alpha: 1),
                width: note.width - (note.isQuote ? 8 : 0),
                focusOnAppear: store.focusNoteID == note.id,
                onChange: { store.setNoteText(note.id, $0) },
                onFocus: { focused in
                    if focused {
                        if store.focusNoteID == note.id { store.focusNoteID = nil }
                        store.beginEditing(note.id)
                    } else if store.editingNote?.id == note.id {
                        store.endEditing()
                    }
                },
                onTab: {
                    // The next line, at the left edge of the page.
                    let rows = NotebookStyle.rows(note.text, width: note.width)
                    let next = note.y + CGFloat(rows) * NotebookStyle.lineHeight
                    store.focusNoteID = store.addNote(at: CGPoint(x: marginX + 4, y: next), pageWidth: pageWidth)
                }
            )
        }
        .fixedSize(horizontal: false, vertical: true) // only as tall as the writing
        .frame(width: note.width, alignment: .leading)
        .background(
            RoundedRectangle(cornerRadius: 4)
                .fill(Color.black.opacity(isFocused ? 0.025 : 0))
                .padding(-4)
        )
        .overlay(alignment: .topLeading) {
            if hovering || isFocused || drag != .zero {
                Image(systemName: "line.3.horizontal")
                    .font(.system(size: 9, weight: .bold))
                    .foregroundStyle(NotebookStyle.ink.opacity(0.3))
                    .frame(width: 16, height: 22)
                    .contentShape(Rectangle())
                    .offset(x: note.isQuote ? -38 : -20, y: 2)
                    .gesture(
                        DragGesture(coordinateSpace: .named("page"))
                            .onChanged { drag = $0.translation }
                            .onEnded { value in
                                store.moveNote(note.id, to: CGPoint(x: note.x + value.translation.width, y: note.y + value.translation.height), pageWidth: pageWidth)
                                drag = .zero
                            }
                    )
                    .help("Drag to move")
            }
        }
        .onHover { hovering = $0 }
        .offset(x: note.x + drag.width, y: note.y + NotebookStyle.textOffset(font) + drag.height)
        .contextMenu {
            Button(note.isQuote ? "Make Normal Text" : "Make Quote") { store.toggleQuote(note.id) }
            Button("Copy") {
                NSPasteboard.general.clearContents()
                NSPasteboard.general.setString(note.text, forType: .string)
            }
            Divider()
            Button("Delete", role: .destructive) { store.deleteNote(note.id) }
        }
    }
}

// MARK: - Notebook writing

/// The text of one note on a notebook page. Every row is exactly one ruled line tall,
/// so writing always sits on the lines. Return makes a new row on the line below;
/// Tab asks for a new note at the left edge of the page.
struct NotebookTextView: NSViewRepresentable {
    let text: String
    let font: NSFont
    let color: NSColor
    let width: CGFloat
    var focusOnAppear = false
    let onChange: (String) -> Void
    let onFocus: (Bool) -> Void
    let onTab: () -> Void

    func makeCoordinator() -> Coordinator { Coordinator(self) }

    func makeNSView(context: Context) -> LinedTextView {
        let view = LinedTextView(frame: NSRect(x: 0, y: 0, width: width, height: NotebookStyle.lineHeight))
        view.isRichText = false
        view.drawsBackground = false
        view.allowsUndo = true
        view.isAutomaticQuoteSubstitutionEnabled = false
        view.textContainerInset = .zero
        view.textContainer?.lineFragmentPadding = 0
        view.textContainer?.widthTracksTextView = true
        view.isVerticallyResizable = true
        view.isHorizontallyResizable = false
        view.delegate = context.coordinator
        view.focusOnAppear = focusOnAppear
        apply(to: view)
        view.string = text
        return view
    }

    func updateNSView(_ view: LinedTextView, context: Context) {
        context.coordinator.parent = self
        apply(to: view)
        if view.string != text, !view.hasMarkedText() { view.string = text }
        view.invalidateIntrinsicContentSize()
    }

    private func apply(to view: LinedTextView) {
        let style = NSMutableParagraphStyle()
        style.minimumLineHeight = NotebookStyle.lineHeight
        style.maximumLineHeight = NotebookStyle.lineHeight
        style.baseWritingDirection = .natural
        let attributes: [NSAttributedString.Key: Any] = [.font: font, .foregroundColor: color, .paragraphStyle: style]
        view.typingAttributes = attributes
        view.defaultParagraphStyle = style
        view.font = font
        view.textColor = color
        if let storage = view.textStorage, storage.length > 0 {
            storage.setAttributes(attributes, range: NSRange(location: 0, length: storage.length))
        }
        view.lineWidth = width
    }

    final class LinedTextView: NSTextView {
        var lineWidth: CGFloat = 200
        var focusOnAppear = false

        /// As tall as its rows; as wide as the note.
        override var intrinsicContentSize: NSSize {
            guard let layout = layoutManager, let container = textContainer else { return NSSize(width: lineWidth, height: NotebookStyle.lineHeight) }
            layout.ensureLayout(for: container)
            let used = layout.usedRect(for: container).height
            return NSSize(width: lineWidth, height: max(NotebookStyle.lineHeight, ceil(used)))
        }

        override func didChangeText() {
            super.didChangeText()
            invalidateIntrinsicContentSize()
        }

        override func viewDidMoveToWindow() {
            super.viewDidMoveToWindow()
            guard focusOnAppear else { return }
            focusOnAppear = false
            DispatchQueue.main.async { [weak self] in
                guard let self, let window = self.window else { return }
                window.makeKey()
                window.makeFirstResponder(self)
            }
        }
    }

    final class Coordinator: NSObject, NSTextViewDelegate {
        var parent: NotebookTextView
        init(_ parent: NotebookTextView) { self.parent = parent }

        func textDidChange(_ notification: Notification) {
            guard let view = notification.object as? NSTextView else { return }
            parent.onChange(view.string)
        }

        func textDidBeginEditing(_ notification: Notification) { parent.onFocus(true) }
        func textDidEndEditing(_ notification: Notification) { parent.onFocus(false) }

        func textView(_ textView: NSTextView, doCommandBy selector: Selector) -> Bool {
            if selector == #selector(NSResponder.insertTab(_:)) {
                parent.onTab()
                return true
            }
            if selector == #selector(NSResponder.cancelOperation(_:)) {
                textView.window?.makeFirstResponder(nil)
                return true
            }
            return false
        }
    }
}
